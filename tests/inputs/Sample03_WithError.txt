// Aggregates and records: ABM with use, const, type record, arrays
use std;
$
const int MAX = 100;
type record Point { int x; int y; }

agent Walker {
    Point pos;
    int trail[5];

    update {
        pos.x = pos.x + 1;
        pos.y = pos.y + 1;
        trail[0] = pos.x;
        move(pos.x, pos.y);
    }
}

world Field {
    int width = MAX;
    int height = MAX;
    int grid[10];

    update {
        grid[0] = 0;
    }
}

void main() {
    Point p;
    p.x = 0;
    p.y = 0;
    int arr[10];
    arr[0] = 1;
    spawn Walker(p, arr) for 3;
    repeat {
        step();
    } until (steps > 50);
}
