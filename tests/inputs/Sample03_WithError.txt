// Predator-prey: Prey move randomly; Predators chase and consume nearby Prey.
agent Prey {
    int x, y;

    update {
        move(self.x + rand(-1, 1), self.y + rand(-1, 1));
    }
}

agent Predator {
    int x, y;
    string status = "hunting
    update {
        targets = neighbors(self, 2);
        if (targets.length > 0) {
            destroy(targets[0]);
        } else {
            move(self.x + rand(-1, 1), self.y + rand(-1, 1));
        }
    }
}

world Meadow {
    int width = 40;
    int height = 40;
    bool spawned = false;

    pre {
        if (!spawned) {
            for (int i = 0; i < 15; i = i + 1) {
                spawn Prey(rand(0, 40), rand(0, 40));
            }
            for (int i = 0; i < 3; i = i + 1) {
                spawn Predator(rand(0, 40), rand(0, 40));
            }
            spawned = true;
        }
    }

    post {
        // tick
    }
}

void main() {
    repeat {
        step();
    } until (steps > 150);
}
