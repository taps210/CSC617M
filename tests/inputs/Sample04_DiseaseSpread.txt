// Disease spread: people move in a city and infect others when close. Population can increase or decrease.
agent Person {
    int x, y;
    int infected;

    update {
        move(self.x + rand(-1, 1), self.y + rand(-1, 1));
        if (infected == 0) {
            nearby = neighbors(self, 2);
            if (nearby.length > 0) {
                infected = 1;
            }
        }
        if (infected == 1 && rand(0, 100) > 95) {
            destroy(self);
        }
    }
}

world City {
    int width = 50;
    int height = 50;
    bool spawned = false;
    int tick = 0;

    pre {
        if (!spawned) {
            for (int i = 0; i < 20; i = i + 1) {
                spawn Person(rand(0, 50), rand(0, 50), 0);
            }
            spawn Person(25, 25, 1);
            spawned = true;
        }
    }

    post {
        tick = tick + 1;
        if (tick % 50 == 0) {
            spawn Person(rand(0, 50), rand(0, 50), 0);
        }
    }
}

void main() {
    repeat {
        step();
    } until (steps > 500);
}
