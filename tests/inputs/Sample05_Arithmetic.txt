agent Trader {
    int x, y;
    int price;
    int quantity;
    float cost;

    update {
        price = price * 2 - 1;
        quantity = quantity / 2;
        cost = cost + 0.5 * 2.0;
        int tax = price % 10;
        move(x + rand(-1, 1), y + rand(0, 2));
        if (quantity * price > 100) {
            quantity = quantity - 1;
        }
        if (cost - 1.0 <= 0.0) {
            destroy(self);
        }
    }
}
