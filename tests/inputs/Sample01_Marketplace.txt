// Marketplace: Trader and Collector in a shared world. Collector drops items; Trader picks them up if nearby (neighbors).
agent Drop {
    int x, y;
    update{}
}

agent Collector {
    int x, y;
    int items = 5;

    update {
        move(self.x + rand(-1, 1), self.y + rand(-1, 1));
        if (items > 0) {
            spawn Drop(self.x, self.y);
            items = items - 1;
        }
    }
}

agent Trader {
    int x, y;
    Drop drops[];

    update {
        move(self.x + rand(-1, 1), self.y + rand(-1, 1));
        drops = neighbors(self, 2);
        if (drops.length > 0) {
            destroy(drops[0]);
        }
    }
}

world Marketplace {
    int width = 20;
    int height = 20;
    bool spawned = false;

    pre {
        if (!spawned) {
            for (int i = 0; i < 3; i = i + 1) {
                spawn Collector(rand(0, 20), rand(0, 20));
            }
            for (int j = 0; j < 2; j = j + 1) {
                spawn Trader(rand(0, 20), rand(0, 20));
            }
            spawned = true;
        }
    }

    post {
        // marketplace tick
    }
}

void main() {
    int steps = 0;
    repeat {
        step();
        steps=steps+1;
    } until (steps > 100);
}
