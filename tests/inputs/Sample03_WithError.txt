// Foragers with position and trail history (records and arrays)
use std;
$
const int MAX = 100;
type record Point { int x; int y; }

agent Forager {
    Point pos;
    int trail[5];

    update {
        pos.x = pos.x + 1;
        pos.y = pos.y + 1;
        trail[0] = pos.x;
        move(pos.x, pos.y);
    }
}

world Nest {
    int width = MAX;
    int height = MAX;
    int grid[10];
    bool spawned = false;

    update {
        if (!spawned) {
            Point p;
            p.x = 0;
            p.y = 0;
            int arr[10];
            arr[0] = 1;
            for (int i = 0; i < 3; i = i + 1) {
                spawn Forager(p, arr);
            }
            spawned = true;
        }
        grid[0] = 0;
    }
}

void main() {
    repeat {
        step();
    } until (steps > 50);
}
