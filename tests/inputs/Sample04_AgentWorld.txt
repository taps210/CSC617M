// Predator-prey ecosystem: sheep and wolves, move, neighbors, spawn, step, destroy
agent Sheep {
    int x, y;
    int energy;

    update {
        move(self.x + rand(-1, 1), self.y + rand(-1, 1));
        energy = energy - 1;

        if (energy <= 0) {
            destroy(self);
        }
    }
}

agent Wolf {
    int x, y;
    int hunger;

    update {
        hunger = hunger + 1;

        sheep = neighbors(self, 3);
        if (sheep.length > 0) {
            destroy(sheep[0]);
            hunger = 0;
        }

        if (hunger > 30) {
            destroy(self);
        }
    }
}

world Ecosystem {
    int width = 100;
    int height = 100;
    bool spawned = false;

    update {
        if (!spawned) {
            for (int i = 0; i < 50; i = i + 1) {
                spawn Sheep(rand(0, 100), rand(0, 100));
            }
            for (int i = 0; i < 10; i = i + 1) {
                spawn Wolf(rand(0, 100), rand(0, 100));
            }
            spawned = true;
        }
        // Environmental updates
    }
}

void main() {
    repeat {
        step();
    } until (steps > 1000);
}
