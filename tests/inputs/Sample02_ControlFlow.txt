// Prey on a pasture: energy-based grazing, movement, death; main sets noise to affect behavior
agent Prey {
    int x, y;
    int energy;

    update {
        if (energy <= 0) {
            destroy(self);
        } else {
            energy = energy - 1;
            if (Pasture.noiseLevel > 0.2) {
                energy = energy - 1;
            }
            move(x + 1, y);
        }
        for (int i = 0; i < 2; i = i + 1) {
            if (energy < 5) { break; }
            continue;
        }
        int dx = (energy > 10 ? 1 : -1);
        move(x + dx, y);
    }
}

world Pasture {
    int width = 20;
    int height = 20;
    float noiseLevel;
    bool spawned = false;

    update {
        if (!spawned) {
            for (int i = 0; i < 5; i = i + 1) {
                spawn Prey(0, 0, 15);
            }
            spawned = true;
        }
        if (width > 0) {
            while (height > 0) {
                height = height - 1;
            }
        }
    }
}

void main() {
    Pasture.noiseLevel = 0.3;
    repeat {
        step();
    } until (steps > 100);
}
