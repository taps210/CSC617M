// Config-driven run: read simulation params, spawn sensor monitors, print status
agent Sensor {
    int id;
    float threshold;
    string name;
    bool active;
    char zone;

    update {
        if (active) {
            move(0, 0);
        }
    }
}

world Env {
    int size = 10;
    bool spawned = false;

    update {
        if (!spawned) {
@
            for (int i = 0; i < 3; i = i + 1) {
                spawn Sensor(1, 0.5, "sensor_A", true, 'A');
            }
            spawned = true;
        }
        // env tick
    }
}

void main() {
    int gridSize = 0;
    float seed = 0.0;
    string runName = "run";
    bool verbose = true;
    char mode = 'N';
    read(gridSize);
    read(seed);
    read(runName);
    read(verbose);
    read(mode);
    print(runName);
    print(gridSize);
    step();
    step();
    print(verbose);
}
