// Resource competition: agents collect resources; resource regrows in world post; temperature influences agent energy loss.
agent Forager {
    int x, y;
    int energy = 100;

    update {
        move(self.x + rand(-1, 1), self.y + rand(-1, 1));
        if (Field.temperature > 5) {
            energy = energy - 2;
        } else {
            energy = energy - 1;
        }
        if (energy <= 0) {
            destroy(self);
        }
    }
}

world Field {
    int width = 40;
    int height = 40;
    int resource = 100;
    int temperature = 3;
    bool spawned = false;

    pre {
        if (!spawned) {
            for (int i = 0; i < 8; i = i + 1) {
                spawn Forager(rand(0, 40), rand(0, 40));
            }
            spawned = true;
        }
    }

    post {
        if (resource < 200) {
            resource = resource + 5;
        }
        temperature = 3 + (steps % 10) - 5;
    }
}

void main() {
    repeat {
        step();
    } until (steps > 300);
}
